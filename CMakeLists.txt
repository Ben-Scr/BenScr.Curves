cmake_minimum_required(VERSION 3.28)
project(BenScrCurves VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
endif()

option(BENSCR_CURVES_BUILD_TESTS "Build contract tests" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_CURVES_BUILD_EXAMPLES "Build example" ${PROJECT_IS_TOP_LEVEL})
option(BENSCR_CURVES_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(benscr_curves INTERFACE)
add_library(BenScr::Curves ALIAS benscr_curves)
set_target_properties(benscr_curves PROPERTIES EXPORT_NAME Curves)
target_compile_features(benscr_curves INTERFACE cxx_std_20)
target_include_directories(benscr_curves INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(benscr_curves_checks target)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(BENSCR_CURVES_SANITIZERS)
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        else()
            message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported POSIX host.")
        endif()
    endif()
endfunction()

if(BUILD_TESTING AND BENSCR_CURVES_BUILD_TESTS)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(curves_contract_tests tests/curves_tests.cpp)
    target_link_libraries(curves_contract_tests PRIVATE BenScr::Curves Threads::Threads)
    benscr_curves_checks(curves_contract_tests)
    add_test(NAME curves_contract COMMAND curves_contract_tests)
    add_executable(curves_header_test tests/header.cpp tests/header_peer.cpp)
    target_link_libraries(curves_header_test PRIVATE BenScr::Curves)
    benscr_curves_checks(curves_header_test)
    add_test(NAME standalone_header COMMAND curves_header_test)
endif()
if(BENSCR_CURVES_BUILD_EXAMPLES)
    add_executable(curves_example examples/curves.cpp)
    target_link_libraries(curves_example PRIVATE BenScr::Curves)
    benscr_curves_checks(curves_example)
endif()

install(TARGETS benscr_curves EXPORT BenScrCurvesTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/BenScrCurves)
configure_package_config_file(cmake/BenScrCurvesConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrCurvesConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrCurves)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/BenScrCurvesConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion ARCH_INDEPENDENT)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BenScrCurvesConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/BenScrCurvesConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrCurves)
install(EXPORT BenScrCurvesTargets NAMESPACE BenScr:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/BenScrCurves)
